## 2. 快速开始

### 2.1 第一个多模态检索

```python
import sys
sys.path.append("src")

from multimodal_retriever import MultimodalRetriever, MultimodalDocument

# 创建检索器
retriever = MultimodalRetriever()

retriever.add_document(MultimodalDocument(content="机器学习是人工智能的分支"))

retriever.add_document(MultimodalDocument(content="自然语言处理研究文本理解"))

# 检索
results = retriever.search("神经网络", top_k=2)
for r in results:
    print(f"[{r.score:.3f}] {r.document.content}")
```

## 3. 多模态检索教程

### 3.1 文本检索

```python
from multimodal_retriever import MultimodalRetriever, MultimodalDocument

# 创建检索器
retriever = MultimodalRetriever()

# 添加知识库
docs = [
    "Python是一种解释型编程语言",
    "JavaScript主要用于Web开发",
    "机器学习需要大量数据",
    "深度学习是机器学习的子集",
]
for text in docs:
    retriever.add_document(MultimodalDocument(content=text))

# 检索
results = retriever.search_by_text("编程语言", top_k=2)
print("检索结果:")
for r in results:
    print(f"  {r.document.content}")
```

### 3.2 图像检索

```python
import numpy as np

# 创建模拟图像
cat_image = np.random.rand(64, 64, 3).astype(np.float32)
dog_image = np.random.rand(64, 64, 3).astype(np.float32)

# 添加图像文档
retriever.add_document(MultimodalDocument(
    content="一只可爱的猫",
    image_data=cat_image,
))
retriever.add_document(MultimodalDocument(
    content="一只活泼的狗",
    image_data=dog_image,
))

# 用文本检索图像
results = retriever.search_by_text("猫", top_k=1)
print(f"找到: {results[0].document.content}")
```

### 3.3 跨模态检索

```python
# 用图像检索相似图像
query_image = np.random.rand(64, 64, 3).astype(np.float32)
results = retriever.search_by_image(query_image, top_k=3)

print("相似图像:")
for r in results:
    print(f"  [{r.score:.3f}] {r.document.content}")
```

## 4. 视觉问答教程

### 4.1 基础问答

```python
from vision_qa_agent import VisionQAAgent

# 创建智能体
agent = VisionQAAgent(max_steps=5)

# 提问
result = agent.answer("什么是机器学习?")
print(f"答案: {result.answer}")
print(f"推理步数: {result.num_steps}")
```

### 4.2 带图像的问答

```python
import numpy as np

# 准备图像
image = np.random.rand(128, 128, 3).astype(np.float32)

# 提问
result = agent.answer("描述这张图片", image=image)
print(f"答案: {result.answer}")

# 查看推理过程
for step in result.steps:
    print(f"Step {step.step_num}:")
    print(f"  思考: {step.thought}")
    print(f"  动作: {step.action.action_type.value}")
```

### 4.3 结合检索的问答

```python
from multimodal_retriever import MultimodalRetriever, MultimodalDocument
from vision_qa_agent import VisionQAAgent

# 创建带知识库的智能体
retriever = MultimodalRetriever()
retriever.add_document(MultimodalDocument(content="猫是一种常见的宠物"))
retriever.add_document(MultimodalDocument(content="狗被称为人类最好的朋友"))

agent = VisionQAAgent(retriever=retriever, max_steps=5)

# 提问
result = agent.answer("常见的宠物有哪些?")
print(f"答案: {result.answer}")
print(f"参考来源: {len(result.sources)}个文档")
```

## 5. 代码助手教程

### 5.1 代码检索

```python
from code_retriever import CodeRetriever, CodeDocument, CodeLanguage

# 创建检索器
retriever = CodeRetriever()

# 添加代码
retriever.add_document(CodeDocument(
    content="def quicksort(arr):\n    if len(arr) <= 1:\n        return arr\n    ...",
    name="quicksort",
    language=CodeLanguage.PYTHON,
))
retriever.add_document(CodeDocument(
    content="def binary_search(arr, target):\n    left, right = 0, len(arr)-1\n    ...",
    name="binary_search",
    language=CodeLanguage.PYTHON,
))

# 检索
results = retriever.search("排序算法", top_k=2)
for r in results:
    print(f"[{r.score:.3f}] {r.document.name}")
```

## 6. 性能测试教程

### 6.1 延迟测试

```python
from rag_benchmark import RAGBenchmark

benchmark = RAGBenchmark()

# 模拟检索函数
def mock_retrieve(query):
    import time
    time.sleep(0.01)
    return ["doc1", "doc2"]

# 运行测试
result = benchmark.run_latency_test(
    mock_retrieve,
    queries=["q1", "q2", "q3"],
    num_runs=5
)

print(f"平均延迟: {result.metrics.retrieval_latency_ms:.2f}ms")
print(f"P95延迟: {result.details['p95_ms']:.2f}ms")
```

### 6.2 准确率测试

```python
# 模拟检索函数
def mock_retrieve(query):
    return ["doc1", "doc2", "doc3"]

# 真实标签
queries = ["q1", "q2"]
ground_truth = [["doc1", "doc2"], ["doc2", "doc3"]]

result = benchmark.run_accuracy_test(mock_retrieve, queries, ground_truth, k=3)
print(f"Recall@3: {result.metrics.recall_at_k:.2%}")
print(f"MRR: {result.metrics.mrr:.3f}")
```

### 6.3 Agent性能测试

```python
from agent_benchmark import AgentBenchmark

benchmark = AgentBenchmark()

def mock_agent(task):
    return {"success": True, "steps": 3, "tool_calls": 2}

result = benchmark.run_task_completion_test(
    mock_agent,
    tasks=["task1", "task2", "task3"]
)

print(f"成功率: {result.metrics.success_rate:.1%}")
print(f"平均步数: {result.metrics.avg_steps:.1f}")
```

## 7. 常见问题

### Q1: 如何提高检索准确率?

1. 增加文档数量
2. 优化文档分块策略
3. 使用更好的嵌入模型

### Q2: Agent推理步数过多怎么办?

1. 减少 `max_steps` 参数
2. 优化提示模板
3. 提供更多上下文

### Q3: 代码审查规则如何自定义?

```python
reviewer = ReviewAgent()
# 添加自定义规则
reviewer.PYTHON_PATTERNS.append(
    (r"print\(", IssueSeverity.INFO, IssueCategory.STYLE, "生产代码避免print")
)
```

### Q4: 如何集成真实的LLM?

```python
def my_llm(prompt):
    # 调用你的LLM API
    response = call_your_llm_api(
        model="your-model",
        messages=[{"role": "user", "content": prompt}]
    )
    return response.content

agent = VisionQAAgent(llm_func=my_llm)
```
